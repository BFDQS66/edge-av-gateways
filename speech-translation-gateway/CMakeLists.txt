cmake_minimum_required(VERSION 3.16)
project(speech_translation_gateway VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(SHERPA_ONNX_ROOT "" CACHE PATH "Root of the sherpa-onnx shared-library package")
if(NOT SHERPA_ONNX_ROOT)
  message(FATAL_ERROR "Set SHERPA_ONNX_ROOT to the extracted sherpa-onnx shared-lib directory")
endif()

find_path(SHERPA_ONNX_INCLUDE_DIR NAMES sherpa-onnx/c-api/c-api.h
          PATHS "${SHERPA_ONNX_ROOT}/include" NO_DEFAULT_PATH)
find_library(SHERPA_ONNX_C_API_LIBRARY NAMES sherpa-onnx-c-api
             PATHS "${SHERPA_ONNX_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT SHERPA_ONNX_INCLUDE_DIR OR NOT SHERPA_ONNX_C_API_LIBRARY)
  message(FATAL_ERROR "Missing sherpa-onnx C header or shared library in SHERPA_ONNX_ROOT")
endif()

add_executable(speech_cli src/speech_cli.cpp)
target_include_directories(speech_cli PRIVATE "${SHERPA_ONNX_INCLUDE_DIR}")
target_link_libraries(speech_cli PRIVATE "${SHERPA_ONNX_C_API_LIBRARY}")
target_compile_options(speech_cli PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(speech_cli PROPERTIES BUILD_RPATH "${SHERPA_ONNX_ROOT}/lib")

find_package(Threads REQUIRED)
set(HTTPLIB_ROOT "${CMAKE_SOURCE_DIR}/deps/cpp-httplib" CACHE PATH "cpp-httplib header directory")
if(EXISTS "${HTTPLIB_ROOT}/httplib.h")
  add_executable(speech_server src/speech_server.cpp)
  target_include_directories(speech_server PRIVATE "${SHERPA_ONNX_INCLUDE_DIR}" "${HTTPLIB_ROOT}")
  target_link_libraries(speech_server PRIVATE "${SHERPA_ONNX_C_API_LIBRARY}" Threads::Threads)
  target_compile_options(speech_server PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(speech_server PROPERTIES BUILD_RPATH "${SHERPA_ONNX_ROOT}/lib")

  add_executable(quality_tts_server src/quality_tts_server.cpp)
  target_include_directories(quality_tts_server PRIVATE "${SHERPA_ONNX_INCLUDE_DIR}" "${HTTPLIB_ROOT}")
  target_link_libraries(quality_tts_server PRIVATE "${SHERPA_ONNX_C_API_LIBRARY}" Threads::Threads)
  target_compile_options(quality_tts_server PRIVATE -Wall -Wextra -Wpedantic)
  set_target_properties(quality_tts_server PROPERTIES BUILD_RPATH "${SHERPA_ONNX_ROOT}/lib")
else()
  message(STATUS "cpp-httplib header not found; run scripts/install_vm_models.sh for speech_server")
endif()

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND AND EXISTS "${HTTPLIB_ROOT}/httplib.h")
  pkg_check_modules(JSONCPP QUIET IMPORTED_TARGET jsoncpp)
  if(JSONCPP_FOUND)
    add_executable(translation_server src/translation_server.cpp)
    target_include_directories(translation_server PRIVATE "${HTTPLIB_ROOT}")
    target_link_libraries(translation_server PRIVATE PkgConfig::JSONCPP Threads::Threads)
    target_compile_options(translation_server PRIVATE -Wall -Wextra -Wpedantic)
  else()
    message(STATUS "jsoncpp not found; translation_server is unavailable")
  endif()
endif()
